// This page reviews payments, approves or rejects bank transfers, and issues refunds.
import { useEffect, useState } from 'react';
import { ExternalLink } from 'lucide-react';
import { adminApi, downloadCsv } from '../api/admin.js';
import { errorMessage } from '../api/client.js';
import { DataTable, FormField, LoadingState, PageHeading, Pagination, Panel, Toast, Button, controlClass, formatPaise, formatDate } from '../components/ui.jsx';
 
const statuses = ['created', 'pending_verification', 'paid', 'failed', 'rejected', 'refunded', 'partially_refunded'];
const initialFilters = { page: 1, limit: 20, status: '', method: '', from: '', to: '' };
 
export default function PaymentsPage() {
  const [filters, setFilters] = useState(initialFilters);
  const [data, setData] = useState(null);
  const [selected, setSelected] = useState(null);
  const [action, setAction] = useState('');
  const [reason, setReason] = useState('');
  const [amount, setAmount] = useState('');
  const [reversePoints, setReversePoints] = useState(false);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  useEffect(() => {
    let active = true;
    setData(null);
    setError('');
    adminApi.payments(filters).then((result) => { Eif (active) setData(result); })
      .catch((failure) => { if (active) setError(errorMessage(failure)); });
    return () => { active = false; };
  }, [filters]);
  async function openDetails(row) {
    try { setSelected(await adminApi.payment(row._id)); }
    catch (failure) { setError(errorMessage(failure)); }
  }
  async function runAction(event) {
    event.preventDefault();
    try {
      const body = action === 'reject' ? { reason }
        : action === 'refund' ? { amount: Math.round(Number(amount) * 100), reason, reversePoints }
          : {};
      await adminApi.paymentAction(selected._id, action, body);
      setAction(''); setReason(''); setNotice('Payment action completed.');
      const fresh = await adminApi.payment(selected._id);
      setSelected(fresh);
      setFilters({ ...filters });
    } catch (failure) { setError(errorMessage(failure)); }
  }
  const columns = [
    { key: 'createdAt', label: 'Date', render: (row) => formatDate(row.createdAt) },
    { key: 'user', label: 'Member', render: (row) => row.user?.firstName || '—' },
    { key: 'plan', label: 'Plan', render: (row) => row.plan?.title || '—' },
    { key: 'method', label: 'Method' }, { key: 'status', label: 'Status' },
    { key: 'totalAmount', label: 'Total', render: (row) => formatPaise(row.totalAmount) },
    { key: 'details', label: '', render: (row) => <Button variant="secondary" onClick={() => openDetails(row)}>Review</Button> },
  ];
  return <section>
    <PageHeading eyebrow="COLLECTIONS" title="Payments" description="Review bank-transfer proofs and track payment decisions." actions={<Button variant="secondary" onClick={() => downloadCsv('/admin/payments/export', filters).catch((failure) => setError(errorMessage(failure)))}>Export CSV</Button>} />
    {error && <Toast kind="error" onDismiss={() => setError('')}>{error}</Toast>}{notice && <Toast onDismiss={() => setNotice('')}>{notice}</Toast>}
    <div className="mb-4 grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
      <select className={controlClass} aria-label="Filter status" value={filters.status} onChange={(event) => setFilters({ ...filters, status: event.target.value, page: 1 })}><option value="">All statuses</option>{statuses.map((status) => <option key={status}>{status}</option>)}</select>
      <select className={controlClass} aria-label="Filter method" value={filters.method} onChange={(event) => setFilters({ ...filters, method: event.target.value, page: 1 })}><option value="">All methods</option><option value="razorpay">Razorpay</option><option value="bank_transfer">Bank transfer</option><option value="manual">Manual</option></select>
      <input className={controlClass} type="date" aria-label="From date" value={filters.from} onChange={(event) => setFilters({ ...filters, from: event.target.value, page: 1 })} />
      <input className={controlClass} type="date" aria-label="To date" value={filters.to} onChange={(event) => setFilters({ ...filters, to: event.target.value, page: 1 })} />
    </div>
    {!data && !error && <LoadingState />}
    {data && <><DataTable columns={columns} rows={data.items} empty="No payment records match these filters." /><Pagination info={data} onPage={(page) => setFilters({ ...filters, page })} /></>}
    {selected && <div className="fixed inset-0 z-40 bg-midnight/40" onClick={() => setSelected(null)} role="presentation"><aside className="absolute inset-y-0 right-0 w-full max-w-xl overflow-y-auto bg-white p-6 shadow-2xl" onClick={(event) => event.stopPropagation()} aria-label="Payment details">
      <button className="float-right text-sm font-semibold text-forest" onClick={() => setSelected(null)}>Close ×</button>
      <p className="mb-1 text-xs font-bold uppercase tracking-widest text-gold">Payment review</p><h2 className="font-display text-2xl">{selected.user?.firstName || 'Member'} · {selected.plan?.title}</h2>
      <p className="my-2 text-sm text-slate-500">{selected.status} · {selected.method}</p>
      <Panel className="my-4 grid gap-2 text-sm"><p>Base {formatPaise(selected.baseAmount)}</p><p>GST {formatPaise(selected.gstAmount)}</p><p>Total <strong>{formatPaise(selected.totalAmount)}</strong></p><p>Invoice: {selected.invoiceNumber || '—'} · UTR: {selected.utr || '—'}</p></Panel>
      {selected.proofUrl && <><a className="inline-flex items-center gap-2 font-semibold text-forest underline" href={`${(import.meta.env.VITE_API_URL || 'http://localhost:5000/api').replace(/\/api$/, '')}${selected.proofUrl}`} target="_blank" rel="noreferrer">View transfer proof <ExternalLink size={15} /></a>{selected.proofUrl.match(/\.(jpg|jpeg|png|webp)$/i) && <img className="mt-3 max-h-72 w-full rounded-xl border border-slate-200 object-contain" src={`${(import.meta.env.VITE_API_URL || 'http://localhost:5000/api').replace(/\/api$/, '')}${selected.proofUrl}`} alt="Uploaded bank-transfer proof" />}</>}
      <div className="mt-5 flex flex-wrap gap-2">
        {selected.method === 'bank_transfer' && selected.status === 'pending_verification' && <><Button onClick={() => setAction('approve')}>Approve transfer</Button><Button variant="danger" onClick={() => setAction('reject')}>Reject transfer</Button></>}
        {selected.method === 'razorpay' && ['paid', 'partially_refunded'].includes(selected.status) && <Button onClick={() => setAction('refund')}>Refund</Button>}
      </div>
    </aside></div>}
    {action && <PaymentActionForm action={action} reason={reason} setReason={setReason} amount={amount} setAmount={setAmount} reversePoints={reversePoints} setReversePoints={setReversePoints} onCancel={() => setAction('')} onSubmit={runAction} />}
  </section>;
}
 
function PaymentActionForm({ action, reason, setReason, amount, setAmount, reversePoints, setReversePoints, onCancel, onSubmit }) {
  return <div className="fixed inset-0 z-[60] grid place-items-center bg-midnight/55 p-4"><form className="w-full max-w-md rounded-2xl bg-white p-6 shadow-xl" onSubmit={onSubmit}>
    <h2 className="mb-4 font-display text-2xl">{action === 'approve' ? 'Approve transfer?' : action === 'reject' ? 'Reject transfer?' : 'Refund payment?'}</h2>
    {action === 'approve' && <p className="mb-4 text-sm text-slate-600">Confirm that this bank transfer has been received and verified.</p>}
    {action !== 'approve' && <FormField label="Reason"><textarea className={controlClass} required value={reason} onChange={(event) => setReason(event.target.value)} /></FormField>}
    {action === 'refund' && <><FormField label="Refund amount (INR)"><input className={controlClass} type="number" min="0.01" step="0.01" required value={amount} onChange={(event) => setAmount(event.target.value)} /></FormField><label className="my-4 flex items-center gap-2 text-sm"><input type="checkbox" checked={reversePoints} onChange={(event) => setReversePoints(event.target.checked)} />Reverse awarded points</label></>}
    <div className="mt-5 flex justify-end gap-2"><Button type="button" variant="secondary" onClick={onCancel}>Cancel</Button><Button type="submit" variant={action === 'reject' ? 'danger' : 'primary'}>{action}</Button></div>
  </form></div>;
}
