      { $sort: { _id: 1 } },
    ]),
    PointsTransaction.aggregate([
      { $match: pointsFilter },
      { $group: { _id: { month: { $dateToString: { format: '%Y-%m', date: '$createdAt' } }, type: '$type' }, total: { $sum: '$amount' } } },
      { $sort: { '_id.month': 1 } },

}
