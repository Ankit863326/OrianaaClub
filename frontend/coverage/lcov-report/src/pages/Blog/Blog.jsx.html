function SlideCapsuleButton({ onExplore }) {
  return (
    <div
      className="b2-card__capsule"
      onClick={(e) => {
        e.stopPropagation();
        onExplore();
      }}
    >
      <Motion.span
        className="b2-card__arrow-circle"
        drag="x"
        dragConstraints={{ left: 0, right: 130 }}
        dragSnapToOrigin={true}
        dragElastic={0.1}
        onDragEnd={(e, info) => {
          if (info.offset.x > 40) {
            onExplore();
          }
        }}
        whileHover={{ scale: 1.1 }}
        whileTap={{ scale: 0.95 }}
      >
        <ChevronRight size={16} />
      </Motion.span>
